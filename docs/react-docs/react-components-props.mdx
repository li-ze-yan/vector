export const title = "Props";
export const description = "Props 是 React 组件间传递数据的机制，实现单向数据流的父到子通信。";

## 概述

Props（Properties）是组件的输入参数，由父组件传入，子组件只读。React 的数据流是**单向的**：从父组件流向子组件。

## Examples

### 基本用法

```tsx
// 定义组件
function Button({ label, onClick }: { label: string; onClick: () => void }) {
  return <button onClick={onClick}>{label}</button>;
}

// 使用组件
function App() {
  return <Button label="Click me" onClick={() => alert("Clicked!")} />;
}
```

### TypeScript 类型定义

```tsx
// interface 定义
interface CardProps {
  title: string;
  description?: string; // 可选
  tags: string[];
  onClose: () => void;
}

function Card({ title, description, tags, onClose }: CardProps) {
  return (
    <div>
      <h2>{title}</h2>
      {description && <p>{description}</p>}
      <div>{tags.map((tag) => <span key={tag}>{tag}</span>)}</div>
      <button onClick={onClose}>Close</button>
    </div>
  );
}
```

### 默认值

```tsx
// 解构时设置默认值
function Avatar({ src, size = 40 }: { src: string; size?: number }) {
  return <img src={src} width={size} height={size} />;
}

// 使用
<Avatar src="/photo.jpg" />        // size = 40
<Avatar src="/photo.jpg" size={80} /> // size = 80
```

### children

`children` 是特殊的 prop，代表组件标签内的内容：

```tsx
function Panel({ title, children }: { title: string; children: React.ReactNode }) {
  return (
    <div className="panel">
      <h3>{title}</h3>
      <div>{children}</div>
    </div>
  );
}

// 使用
<Panel title="Settings">
  <p>This is the panel content.</p>
  <button>Save</button>
</Panel>
```

### 展开传递

```tsx
interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label: string;
}

function LabeledInput({ label, ...rest }: InputProps) {
  return (
    <label>
      {label}
      <input {...rest} />
    </label>
  );
}

// 所有原生 input 属性都会传递
<LabeledInput label="Email" type="email" placeholder="you@example.com" required />
```

### 回调函数 — 子到父通信

```tsx
function SearchInput({ onSearch }: { onSearch: (query: string) => void }) {
  const [value, setValue] = useState("");

  return (
    <form onSubmit={(e) => { e.preventDefault(); onSearch(value); }}>
      <input value={value} onChange={(e) => setValue(e.target.value)} />
      <button type="submit">Search</button>
    </form>
  );
}

function App() {
  const handleSearch = (query: string) => {
    console.log("Searching:", query);
  };

  return <SearchInput onSearch={handleSearch} />;
}
```
