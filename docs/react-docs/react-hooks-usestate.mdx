export const title = "useState";
export const description = "useState 是最基础的 React Hook，用于在函数组件中添加和管理局部状态。";

## 基本用法

```tsx
const [state, setState] = useState(initialValue);
```

`useState` 返回一个包含两个元素的数组：
- `state` — 当前状态值
- `setState` — 更新状态的函数

## Examples

### 基础计数器

```tsx
function Counter() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>+1</button>
      <button onClick={() => setCount(count - 1)}>-1</button>
    </div>
  );
}
```

### 函数式更新

当新状态依赖旧状态时，传入函数而非值：

```tsx
// 错误 — 连续调用可能基于同一个旧值
setCount(count + 1);
setCount(count + 1); // count 只会 +1，不是 +2

// 正确 — 函数式更新，基于最新值
setCount((prev) => prev + 1);
setCount((prev) => prev + 1); // count 会 +2
```

### 对象状态

更新对象状态时必须创建新引用：

```tsx
const [user, setUser] = useState({ name: "Alice", age: 25 });

// 错误 — 直接修改
user.name = "Bob"; // React 不会重新渲染

// 正确 — 展开运算符创建新对象
setUser({ ...user, name: "Bob" });
setUser((prev) => ({ ...prev, name: "Bob" }));
```

### 惰性初始化

初始值计算开销大时，传入函数：

```tsx
// 每次渲染都会执行 JSON.parse（浪费）
const [data, setData] = useState(JSON.parse(localStorage.getItem("data") || "{}"));

// 只在首次渲染时执行
const [data, setData] = useState(() => JSON.parse(localStorage.getItem("data") || "{}"));
```

## 类型定义

```tsx
// 自动推导
const [count, setCount] = useState(0); // number

// 显式指定类型
const [user, setUser] = useState<User | null>(null);

// 联合类型
const [status, setStatus] = useState<"idle" | "loading" | "error">("idle");
```
